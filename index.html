Python FastAPI Production Script

Plug this full backend script into your Codespace `main.py` file to serve REST API endpoints for this dashboard.

from fastapi import FastAPI, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import List, Optional
import sqlite3
import requests
import json

app = FastAPI(title="eBay Automation Engine API")

# Database setup helper
def get_db_connection():
    conn = sqlite3.connect("catalog.db")
    conn.row_factory = sqlite3.Row
    return conn

class ProductSchema(BaseModel):
    sku: str
    title: str
    description: Optional[str] = ""
    price: float
    quantity: int
    image_url: Optional[str] = ""

@app.get("/api/products")
def list_products():
    conn = get_db_connection()
    products = conn.execute("SELECT * FROM products").fetchall()
    conn.close()
    return [dict(p) for p in products]

@app.post("/api/products")
def create_product(prod: ProductSchema):
    conn = get_db_connection()
    try:
        conn.execute(
            "INSERT INTO products (sku, title, description, price, quantity, image_url, status) VALUES (?, ?, ?, ?, ?, ?, 'draft')",
            (prod.sku, prod.title, prod.description, prod.price, prod.quantity, prod.image_url)
        )
        conn.commit()
    except sqlite3.IntegrityError:
        raise HTTPException(status_code=400, detail="SKU already exists")
    finally:
        conn.close()
    return {"status": "success", "message": f"Product {prod.sku} created"}

@app.post("/api/validate")
def validate_catalog():
    conn = get_db_connection()
    products = conn.execute("SELECT * FROM products").fetchall()
    results = {"valid_count": 0, "invalid_count": 0}

    for prod in products:
        issues = []
        if len(prod["title"]) > 80:
            issues.append("Title exceeds 80 chars")
        if prod["price"] <= 0:
            issues.append("Price must be > 0")
        if prod["quantity"] < 1:
            issues.append("Quantity must be >= 1")

        new_status = "invalid" if issues else "ready"
        conn.execute("UPDATE products SET status = ? WHERE sku = ?", (new_status, prod["sku"]))
        if issues:
            results["invalid_count"] += 1
        else:
            results["valid_count"] += 1

    conn.commit()
    conn.close()
    return results

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
